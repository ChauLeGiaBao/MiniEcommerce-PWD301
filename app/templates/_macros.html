{# Hiển thị ảnh sản phẩm. - HINHANH là link http(s) -> dùng link đó. - HINHANH là tên file ảnh cục bộ (vd "iphone-14.jpg") -> load từ static/img/products/. - HINHANH rỗng / "no-image.jpg" -> dùng ảnh mặc định no-image.svg. - Link bị hỏng (404, chặn hotlink...) -> onerror tự đổi sang ảnh mặc định. #} {% macro product_image(product, css_class="", height="") %} {% set fallback = url_for('static', filename='img/no-image.svg') %} {% set raw = (product.HINHANH or '').strip() %} {% if raw.startswith('http://') or raw.startswith('https://') %} {% set src = raw %} {% elif raw.startswith('/') %} {% set src = raw %} {% elif raw and raw != 'no-image.jpg' %} {% set src = url_for('static', filename='img/products/' ~ raw) %} {% else %} {% set src = fallback %} {% endif %} {% endmacro %}
